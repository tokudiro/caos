# Class diagram

Classes of caos. `Caos`, `ListBuf`, `Element`, `SBuf`, `RBuf` and `SLib` are written in `src/*.caos` and generated by caos itself (see "Self-hosting" in the [README](../README.md)). `main.c` and the lexer are written in C / flex.

```mermaid
classDiagram
    direction LR

    class Caos {
        <<singleton>>
        -ListBuf* list
        +getInstance()$ Caos*
        +init()
        +list() ListBuf*
    }

    class ListBuf {
        <<singleton>>
        -int length
        -Element** elements
        +getInstance()$ ListBuf*
        +init()
        +append(Element* element)
        +length() int
        +element(int index) Element*
        +finish()
    }

    class Element {
        <<struct>>
        +char* string
        +int length
        +int type
        +new()$ Element*
        +delete()
        +init(const char* string, int length, int type)
        +string() char*
        +finish()
    }

    class SBuf {
        -char str[MAX_TEXT]
        -boolean exist
        -int len
        +init()
        +setBuf(const char* top, const int len)
        +getStr() const char*
        +getLen() int
    }

    class RBuf {
        <<ring buffer>>
        -char buf[MAX_ARRAYSIZE][MAX_TEXT]
        -buftype type[MAX_ARRAYSIZE]
        -char thispointer_str[MAX_TEXT]
        -char thisclass_str[MAX_TEXT]
        -boolean isVerbose
        -int first
        -int last
        +init(const char* thispointer_str, boolean isVerbose)
        +setClass(const char* thisclass_str)
        +enque(const char* str, const int len, const buftype type)
        +deque(buftype* p_type) char*
        +isEmpty() boolean
        +allDeque(char* buf) char*
        +back_retype(const buftype find, const buftype replace) boolean
        +back_getStr(const buftype find) const char*
        +trimQue()
        +isWSPACE() boolean
        +toString(char* tostring) char*
    }

    class SLib {
        <<utility>>
        +chopRet(char* buf)$ char*
        +toupper(char* output, const char* input)$
        +checkLength(char* buf)$ int
    }

    class buftype {
        <<enumeration>>
        T_NON
        T_NUMBER
        T_WORD
        T_OBJECT
        T_CLASS
        T_DOT
        T_KEYWORD
        T_METHOD
        T_PARAMETER
        T_ASTERISK
        T_POINTER
        T_TYPE
        T_VALUE
        T_SEMICOLON
        T_WSPACE
        T_COMMA
    }

    class Main["main.c"] {
        +main(int argc, char** argv) int
        -outputHELP()$
        -makePath(char* path, const char* dir, const char* suffix)$
    }

    class Lexer["lexer (flex)"] {
        INITIAL / IMPORT / CLASS_DEFINE
        CLASS_SCOPE / METHOD_DEFINE
        METHOD_SCOPE / ATTRIBUTE_DEFINE
        +yylex() int
    }

    class Globals["global.h"] {
        <<globals>>
        FILE* source
        FILE* public_header
        FILE* private_header
        FILE* define_header
        FILE* struct_header
        RBuf* queue
        SBuf* class_buf
        SBuf* thispointer_buf
        Caos* caos
    }

    Caos "1" o-- "1" ListBuf : list
    ListBuf "1" o-- "0..*" Element : elements
    RBuf ..> buftype : type

    Globals o-- RBuf : queue
    Globals o-- SBuf : class_buf, thispointer_buf
    Globals o-- Caos : caos

    Main ..> Lexer : yylex
    Main ..> Globals
    Main ..> SLib
    Main ..> Caos : init
    Main ..> ListBuf : finish
    Lexer ..> Globals
    Lexer ..> SLib
    Lexer ..> Element : new
    Lexer ..> ListBuf : append
```
